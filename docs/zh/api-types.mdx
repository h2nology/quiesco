### `ElementQuery` {#elementquery}

```ts
interface ElementQuery {
  /** testID：完全一致的字符串或 RegExp。 */
  id?: TextQuery;
  /** 元素子树的文本（返回匹配的最深层元素）。 */
  text?: TextQuery;
  /** accessibilityLabel / aria-label。 */
  label?: TextQuery;
  /** accessibilityRole / role。 */
  role?: string;
  /** 语义类型（`'button'`、`'input'`、`'switch'` …：在所有平台上相同）、宿主组件类型（`'RCTView'`、`'div'`），或渲染该元素的组件的 displayName。 */
  type?: SemanticType | (string & {});
  /** 某个后代元素匹配（Detox 的 `withDescendant`）。 */
  has?: ElementQuery | string;
  /** 元素自身不匹配。 */
  not?: ElementQuery | string;
  /** 位于该匹配器匹配到的唯一元素内部（Detox 的 `withAncestor`）。 */
  within?: ElementQuery | string;
  /** 第 n 个匹配（从 0 开始），按渲染顺序，在所有平台上顺序相同。 */
  index?: number;
}
```

interface RegisterOptions extends LifecycleOptions {
